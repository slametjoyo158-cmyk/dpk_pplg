ProgramPenjualanLaptop.java JDK 17 Compatible 
import java.util.Scanner;

public class ProgramPenjualanLaptop {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        // 1. Input Data
        System.out.print("Masukkan Nama Laptop: ");
        String namaLaptop = input.nextLine();

        System.out.print("Masukkan Harga Satuan: ");
        double harga = input.nextDouble();

        System.out.print("Masukkan Jumlah Beli: ");
        int jumlah = input.nextInt();

        // 2. Hitung Total & Diskon
        double totalBelanja = harga * jumlah;
        double diskon = 0;

        if (totalBelanja >= 10000000) {
            diskon = 0.10 * totalBelanja;
        } else {
            diskon = 0;
        }

        double totalBayar = totalBelanja - diskon;

        // 3. Output Nota
        System.out.println("--- NOTA PENJUALAN ---");
        System.out.println("Laptop: " + namaLaptop);
        System.out.println("Total Belanja: Rp " + totalBelanja);
        System.out.println("Potongan Diskon: Rp " + diskon);
        System.out.println("Total Bayar: Rp " + totalBayar);
    }
}
