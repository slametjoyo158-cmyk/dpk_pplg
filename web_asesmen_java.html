package Latihan_input_Output;
import java.util.*;

/**
 *
 * @author WINDOWS
 */
public class Data_Siswa {

    public static void main(String[] args) {
        // TODO code application logic here
        Scanner input = new Scanner(System.in);

        //1. membuat variabel nama,nis,kelas
        String nama;
        int nis;
        String Kelas;

        //2.input nama
        System.out.print("masukan nama anda :");
        nama=input.nextLine();

        //3. input nis
        System.out.print("masukan NIS anda :");
        nis=input.nextInt();

        //input NextLine setelah nextInt untuk mencegah kursor loncat ke selesai.
        input.nextLine();

        //4. input kelas
        System.out.print("masukan NIS anda :");
        Kelas=input.nextLine();

        //5. Output Tampilkan Data Siswa
        System.out.println("Informasi Data Siswa");
        System.out.println("Nama anda adalah " +nama);
        System.out.println("NIS anda adalah " +nis);
        System.out.println("Kelas anda adalah " +Kelas);

    }//Akhir main
}//Akhir class Data_Siswa
